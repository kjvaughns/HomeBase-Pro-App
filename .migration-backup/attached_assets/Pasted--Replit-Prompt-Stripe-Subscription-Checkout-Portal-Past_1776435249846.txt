## STEP 2 — Add a provider-specific subscription checkout endpoint

File: `server/routes.ts`

Find the existing `/api/stripe/create-checkout-session` route (around line 5866). Read what it currently does. Then add a NEW dedicated subscription endpoint right below it:

```typescript
// POST /api/subscriptions/create-checkout
// Creates a Stripe Checkout session for provider monthly subscription
app.post('/api/subscriptions/create-checkout', requireAuth, async (req, res) => {
  try {
    const userId = (req as any).userId;
    const { providerId } = req.body;

    if (!providerId) {
      return res.status(400).json({ error: 'providerId is required' });
    }

    // Verify the provider belongs to this user
    const provider = await storage.getProviderByUserId(userId);
    if (!provider || provider.id !== providerId) {
      return res.status(403).json({ error: 'Unauthorized' });
    }

    const priceId = process.env.STRIPE_SUBSCRIPTION_PRICE_ID;
    if (!priceId) {
      return res.status(500).json({ error: 'Subscription pricing not configured' });
    }

    // Get or create Stripe customer for this user
    const user = await storage.getUser(userId);
    if (!user) return res.status(404).json({ error: 'User not found' });

    const stripeClient = await getUncachableStripeClient();
    
    // Look up existing customer or create one
    let customerId: string | undefined;
    if (user.stripeCustomerId) {
      customerId = user.stripeCustomerId;
    } else {
      const customer = await stripeClient.customers.create({
        email: user.email,
        name: `${user.firstName} ${user.lastName}`.trim(),
        metadata: { userId, providerId }
      });
      customerId = customer.id;
      // Save customer ID to user record
      await storage.updateUser(userId, { stripeCustomerId: customerId });
    }

    // Create Stripe Checkout session for subscription
    const session = await stripeClient.checkout.sessions.create({
      customer: customerId,
      payment_method_types: ['card'],
      mode: 'subscription',
      line_items: [
        {
          price: priceId,
          quantity: 1,
        },
      ],
      success_url: `https://homebaseproapp.com?subscribed=true`,
      cancel_url: `https://homebaseproapp.com?subscribed=false`,
      metadata: {
        providerId,
        userId,
        type: 'homebase_subscription'
      },
      subscription_data: {
        metadata: {
          providerId,
          userId,
          type: 'homebase_subscription'
        }
      },
      allow_promotion_codes: true,
    });

    return res.json({ url: session.url, sessionId: session.id });
  } catch (err: any) {
    console.error('Subscription checkout error:', err);
    return res.status(500).json({ error: 'Failed to create checkout session', details: err.message });
  }
});
```

```typescript
// POST /api/subscriptions/portal
// Creates a Stripe Customer Portal session for managing subscription
app.post('/api/subscriptions/portal', requireAuth, async (req, res) => {
  try {
    const userId = (req as any).userId;
    const user = await storage.getUser(userId);
    
    if (!user?.stripeCustomerId) {
      return res.status(400).json({ error: 'No subscription found. Please subscribe first.' });

    const stripeClient = await getUncachableStripeClient();
    
    const session = await stripeClient.billingPortal.sessions.create({
      customer: user.stripeCustomerId,
      return_url: `https://homebaseproapp.com`,
    });

    return res.json({ url: session.url });
  } catch (err: any) {
    console.error('Customer portal error:', err);
    return res.status(500).json({ error: 'Failed to open billing portal', details: err.message });
  }
});
```

```typescript
async function handleSubscriptionCancelled(subscription: Stripe.Subscription) {
  try {
    const providerId = subscription.metadata?.providerId;
    if (!providerId) return;
    
    await db.update(providerPlans)
      .set({ isSubscribed: false, planTier: 'free', updatedAt: new Date() })
      .where(eq(providerPlans.providerId, providerId));
    
    console.log(`[Subscription] Provider ${providerId} subscription cancelled`);
    
    const userId = subscription.metadata?.userId;
    if (userId) {
      await sendPush(userId, {
        title: "Subscription cancelled",
        body: "Your HomeBase subscription has been cancelled. You can resubscribe anytime at homebaseproapp.com.",
        data: { type: "subscription_cancelled", screen: "Subscription" }
      }).catch(() => {});
    }
  } catch (err) {
    console.error('[Subscription] Error handling cancellation:', err);
  }
}

async function handleSubscriptionUpdated(subscription: Stripe.Subscription) {
  try {
    const providerId = subscription.metadata?.providerId;
    if (!providerId) return;
    
    const isActive = subscription.status === 'active' || subscription.status === 'trialing';
    
    await db.update(providerPlans)
      .set({ 
        isSubscribed: isActive, 
        planTier: isActive ? 'professional' : 'free',
        updatedAt: new Date()
      })
      .where(eq(providerPlans.providerId, providerId));
    
    console.log(`[Subscription] Provider ${providerId} subscription updated — status: ${subscription.status}`);
  } catch (err) {
    console.error('[Subscription] Error handling subscription update:', err);
  }
}

async function handleSubscriptionPaymentFailed(invoice: Stripe.Invoice) {
  try {
    // Only handle subscription invoices (not regular HomeBase invoices)
    if (!invoice.subscription) return;
    
    const subscription = await (await getUncachableStripeClient()).subscriptions.retrieve(
      invoice.subscription as string
    );
    
    const providerId = subscription.metadata?.providerId;
    const userId = subscription.metadata?.userId;
    
    if (!providerId) return;
    
    console.log(`[Subscription] Payment failed for provider ${providerId}`);
    
    if (userId) {
      await sendPush(userId, {
        title: "Payment failed",
        body: "Your HomeBase subscription payment failed. Update your billing info at homebaseproapp.com.",
        data: { type: "subscription_payment_failed", screen: "Subscription" }
      }).catch(() => {});
    }
  } catch (err) {
    console.error('[Subscription] Error handling payment failure:', err);
  }
}
```

```typescript
import { AppState } from 'react-native';
import { useEffect, useRef } from 'react';

// Inside the useSubscriptionStatus hook, add:
const appState = useRef(AppState.currentState);

useEffect(() => {
  const subscription = AppState.addEventListener('change', nextAppState => {
    if (appState.current.match(/inactive|background/) && nextAppState === 'active') {
      // App came to foreground — refetch subscription status in case they just subscribed
      query.refetch();
    }
    appState.current = nextAppState;
  });

  return () => subscription.remove();
}, []);
```
