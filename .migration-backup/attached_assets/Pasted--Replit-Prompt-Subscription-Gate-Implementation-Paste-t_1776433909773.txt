## STEP 2 — Add a subscription status helper function to server

File: `server/storage.ts` (or create `server/subscriptionUtils.ts`)

Add this helper function that computes a provider's current subscription state:

```typescript
export type SubscriptionStatus = 'free' | 'grace_period' | 'expired' | 'subscribed';

export function computeSubscriptionStatus(plan: {
  isSubscribed: boolean;
  firstPaidBookingAt: Date | null;
  gracePeriodEndsAt: Date | null;
}): {
  status: SubscriptionStatus;
  daysRemainingInGrace: number | null;
  firstPaidBookingAt: Date | null;
  gracePeriodEndsAt: Date | null;
} {
  if (plan.isSubscribed) {
    return { status: 'subscribed', daysRemainingInGrace: null, firstPaidBookingAt: plan.firstPaidBookingAt, gracePeriodEndsAt: plan.gracePeriodEndsAt };
  }
  if (!plan.firstPaidBookingAt) {
    return { status: 'free', daysRemainingInGrace: null, firstPaidBookingAt: null, gracePeriodEndsAt: null };
  }
  const now = new Date();
  if (plan.gracePeriodEndsAt && now < plan.gracePeriodEndsAt) {
    const msRemaining = plan.gracePeriodEndsAt.getTime() - now.getTime();
    const daysRemaining = Math.ceil(msRemaining / (1000 * 60 * 60 * 24));
    return { status: 'grace_period', daysRemainingInGrace: daysRemaining, firstPaidBookingAt: plan.firstPaidBookingAt, gracePeriodEndsAt: plan.gracePeriodEndsAt };
  }
  return { status: 'expired', daysRemainingInGrace: 0, firstPaidBookingAt: plan.firstPaidBookingAt, gracePeriodEndsAt: plan.gracePeriodEndsAt };
}
```

## STEP 3 — Trigger the grace period when first invoice is paid

File: `server/routes.ts`

Find the `POST /api/invoices/:id/mark-paid` endpoint (around line 5580). After the invoice is marked paid and the notification is dispatched, add this logic:

```typescript
// After marking invoice paid — check if this is provider's first paid booking
try {
  const invoice = await storage.getInvoice(invoiceId);
  if (invoice?.providerId) {
    // Get or create provider plan
    let plan = await storage.getProviderPlan(invoice.providerId);
    if (!plan) {
      await storage.createProviderPlan({ providerId: invoice.providerId });
      plan = await storage.getProviderPlan(invoice.providerId);
    }
    
    // Only trigger if this is their first paid booking
    if (plan && !plan.firstPaidBookingAt && !plan.isSubscribed) {
      const now = new Date();
      const gracePeriodEndsAt = new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000); // 7 days
      
      await db.update(providerPlans)
        .set({ 
          firstPaidBookingAt: now,
          gracePeriodEndsAt: gracePeriodEndsAt
        })
        .where(eq(providerPlans.providerId, invoice.providerId));
      
      // Get provider's userId for notifications
      const provider = await storage.getProviderById(invoice.providerId);
      if (provider?.userId) {
        // Send push notification
        await sendPush(provider.userId, {
          title: "You got your first paid booking! 🎉",
          body: "You have 7 days to subscribe at homebaseproapp.com to keep your business running.",
          data: { type: "subscription_grace_start", screen: "Subscription" }
        }).catch(() => {});
        
        // Send email
        await dispatch('subscription.grace_start', {
          recipientUserId: provider.userId,
          recipientEmail: provider.email || '',
          data: {
            providerName: provider.businessName,
            gracePeriodEndsAt: gracePeriodEndsAt.toISOString(),
            subscribeUrl: 'https://homebaseproapp.com/subscribe'
          }
        }).catch(() => {});
      }
    }
  }
} catch (err) {
  console.error('Error triggering subscription grace period:', err);
  // Non-fatal — don't fail the invoice payment
}
```

## STEP 4 — Add scheduled reminders for grace period

// Grace period reminder — day 5 (2 days left warning)
cron.schedule('0 9 * * *', async () => {
  try {
    const twoDaysFromNow = new Date(Date.now() + 2 * 24 * 60 * 60 * 1000);
    const twoDaysFromNowEnd = new Date(twoDaysFromNow.getTime() + 24 * 60 * 60 * 1000);
    
    // Find providers whose grace period ends in ~2 days and are not subscribed
    const expiringProviders = await db
      .select()
      .from(providerPlans)
      .where(
        and(
          gte(providerPlans.gracePeriodEndsAt, twoDaysFromNow),
          lte(providerPlans.gracePeriodEndsAt, twoDaysFromNowEnd),
          eq(providerPlans.isSubscribed, false)
        )
      );
    
    for (const plan of expiringProviders) {
      const provider = await storage.getProviderById(plan.providerId);
      if (provider?.userId) {
        const alreadySent = await hasDeliveryForRecord('subscription.grace_reminder', plan.id);
        if (!alreadySent) {
          await sendPush(provider.userId, {
            title: "2 days left on your free period",
            body: "Subscribe at homebaseproapp.com to keep booking and invoicing clients.",
            data: { type: "subscription_reminder", screen: "Subscription" }
          }).catch(() => {});
          
          await dispatch('subscription.grace_reminder', {
            recipientUserId: provider.userId,
            recipientEmail: provider.email || '',
            relatedRecordType: 'provider_plan',
            relatedRecordId: plan.id,
            data: {
              providerName: provider.businessName,
              daysRemaining: 2,
              subscribeUrl: 'https://homebaseproapp.com/subscribe'
            }
          }).catch(() => {});
        }
      }
    }
  } catch (err) {
    console.error('Grace period reminder cron error:', err);
  }
});

// Grace period expired notification
cron.schedule('0 10 * * *', async () => {
  try {
    const now = new Date();
    
    // Find providers whose grace period just expired (within last 24h) and are not subscribed
    const oneDayAgo = new Date(now.getTime() - 24 * 60 * 60 * 1000);
    
    const expiredProviders = await db
      .select()
      .from(providerPlans)
      .where(
        and(
          lte(providerPlans.gracePeriodEndsAt, now),
          gte(providerPlans.gracePeriodEndsAt, oneDayAgo),
          eq(providerPlans.isSubscribed, false)
        )
      );
    
    for (const plan of expiredProviders) {
      const provider = await storage.getProviderById(plan.providerId);
      if (provider?.userId) {
        const alreadySent = await hasDeliveryForRecord('subscription.grace_expired', plan.id);

          await sendPush(provider.userId, {
            title: "Your free period has ended",

            data: { type: "subscription_expired", screen: "Subscription" }

          await dispatch('subscription.grace_expired', {
            recipientUserId: provider.userId,
            recipientEmail: provider.email || '',
            relatedRecordType: 'provider_plan',
            relatedRecordId: plan.id,
            data: {
              providerName: provider.businessName,
              subscribeUrl: 'https://homebaseproapp.com/subscribe'
            }
          }).catch(() => {});
        }
      }
    }
  } catch (err) {
    console.error('Grace period expired cron error:', err);
  }
});

## STEP 5 — Add subscription status API endpoint

File: `server/routes.ts`

Add this new endpoint near the existing plan endpoints (around line 6356):

```typescript
// GET provider subscription status
app.get('/api/providers/:providerId/subscription-status', requireAuth, async (req, res) => {
  try {
    const { providerId } = req.params;
    
    let plan = await storage.getProviderPlan(providerId);
    if (!plan) {
      // No plan record yet = still free
      return res.json({
        status: 'free',
        daysRemainingInGrace: null,
        firstPaidBookingAt: null,
        gracePeriodEndsAt: null,
        isSubscribed: false
      });
    }
    
    const result = computeSubscriptionStatus(plan);
    return res.json(result);
  } catch (err) {
    console.error('Error fetching subscription status:', err);
    return res.status(500).json({ error: 'Failed to fetch subscription status' });
  }
});

// POST mark provider as subscribed (called when they subscribe via website and return to app)
app.post('/api/providers/:providerId/activate-subscription', requireAuth, async (req, res) => {
  try {
    const { providerId } = req.params;
    
    await db.update(providerPlans)
      .set({ isSubscribed: true, planTier: 'professional' })
      .where(eq(providerPlans.providerId, providerId));
    
    return res.json({ success: true });
  } catch (err) {
    console.error('Error activating subscription:', err);
    return res.status(500).json({ error: 'Failed to activate subscription' });
  }
});
```

## STEP 6 — Add subscription gate middleware for gated routes

File: `server/routes.ts`

Create a helper function near the top of routes.ts:

```typescript
async function checkSubscriptionGate(providerId: string): Promise<{ blocked: boolean; reason: string }> {
  try {
    const plan = await storage.getProviderPlan(providerId);
    if (!plan) return { blocked: false, reason: '' }; // No plan = still free
    
    const { status } = computeSubscriptionStatus(plan);
    
    if (status === 'expired') {
      return { 
        blocked: true, 
        reason: 'Your free period has ended. Subscribe at homebaseproapp.com to continue.' 
      };
    }
    
    return { blocked: false, reason: '' };
  } catch {
    return { blocked: false, reason: '' }; // Never block on error
  }
}
```

Then add the gate check to these three endpoints:

## STEP 7 — Create the useSubscriptionStatus hook

File: `client/hooks/useSubscriptionStatus.ts` (create new file)

```typescript
import { useQuery } from '@tanstack/react-query';
import { useAuthStore } from '../state/authStore';
import { apiRequest } from '../lib/queryClient';

export type SubscriptionStatus = 'free' | 'grace_period' | 'expired' | 'subscribed';

export interface SubscriptionStatusData {
  status: SubscriptionStatus;
  daysRemainingInGrace: number | null;
  firstPaidBookingAt: string | null;
  gracePeriodEndsAt: string | null;
  isSubscribed: boolean;
}

export function useSubscriptionStatus() {
  const { providerProfile } = useAuthStore();
  const providerId = providerProfile?.id;

  const query = useQuery<SubscriptionStatusData>({
    queryKey: ['/api/providers', providerId, 'subscription-status'],
    queryFn: () => apiRequest(`/api/providers/${providerId}/subscription-status`),
    enabled: !!providerId,
    staleTime: 1000 * 60 * 5, // Cache for 5 minutes
    refetchOnWindowFocus: true,
  });

  const isGated = query.data?.status === 'expired';
  const isInGrace = query.data?.status === 'grace_period';
  const isFree = query.data?.status === 'free';
  const isSubscribed = query.data?.status === 'subscribed';

  return {
    ...query,
    status: query.data?.status ?? 'free',
    daysRemainingInGrace: query.data?.daysRemainingInGrace ?? null,
    isGated,
    isInGrace,
    isFree,
    isSubscribed,
  };
}
```
